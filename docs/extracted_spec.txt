sum(expense)- Monthly Income = sum(income)- Category spent = sum(expense where category)6.4 Logic tính toán từ Db6.4.1 Monthly ExpenseSELECT SUM(amount)FROM transactionsWHERE user_id = ?  AND month_key = '2025-12'  AND type = 'expense';6.4.2 Budget Used per CategorySELECT  b.category_id,  b.amount AS budget,  ABS(SUM(t.amount)) AS spent,  ABS(SUM(t.amount)) / b.amount * 100 AS percent_usedFROM budgets bLEFT JOIN transactions t  ON t.category_id = b.category_id AND t.month_key = b.month_key AND t.type = 'expense'WHERE b.user_id = ?  AND b.month_key = '2025-12'GROUP BY b.category_id, b.amount;6.4.3 Cashflow theo ngàySELECT  transaction_date,  SUM(amount) AS cashflowFROM transactionsWHERE user_id = ?  AND month_key = '2025-12'GROUP BY transaction_dateORDER BY transaction_date;6.4.4 Spending Habits (Analytics)SELECT  EXTRACT(DOW FROM transaction_date) AS weekday,  ABS(SUM(amount)) AS totalFROM transactionsWHERE user_id = ?  AND type = 'expense'GROUP BY weekday;6.5 Index đề xuấtCREATE INDEX idx_tx_user_monthON transactions(user_id, month_key);CREATE INDEX idx_tx_user_dateON transactions(user_id, transaction_date);CREATE INDEX idx_budget_user_monthON budgets(user_id, month_key);6.5 Category Data Contract6.5.1 Category Typepublic enum CategoryType {    EXPENSE,    INCOME}6.5.2 Expense Categorypublic enum ExpenseCategory {    FOOD("Food", "🍽", "#4F46E5"),    TRANSPORT("Transport", "🚗", "#6366F1"),    HOUSING("Housing", "🏠", "#8B5CF6"),    BILLS("Bills", "📄", "#7C3AED"),    SHOPPING("Shopping", "🛍", "#EC4899"),    ENTERTAINMENT("Entertainment", "🎬", "#F59E0B"),    COFFEE("Coffee", "☕", "#A16207"),    HEALTHCARE("Healthcare", "💊", "#22C55E"),    EDUCATION("Education", "🎓", "#0EA5E9"),    OTHER("Other", "📦", "#6B7280");     private final String displayName;    private final String icon;    private final String color;}6.5.3 Income Category public enum IncomeCategory {    SALARY("Salary", "💼", "#22C55E"),    FREELANCE("Freelance", "🧑‍💻", "#10B981"),    AFFILIATE("Affiliate", "🔗", "#14B8A6"),    SELLING("Selling", "💸", "#16A34A"),    OTHER_INCOME("Other Income", "➕", "#4ADE80");     private final String displayName;    private final String icon;    private final String color;}6.5.4 Category RulesTransaction Expense chỉ dùng ExpenseCategoryTransaction Income chỉ dùng IncomeCategoryKhông cho xóa category đã có transactionCategory chỉ được disable, không deleteIcon & color là cố định, không đổi runtime7.
